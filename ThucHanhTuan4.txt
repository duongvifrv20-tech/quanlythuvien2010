import java.util.Locale;
import java.util.Scanner;

public class ThucHanhTuan4 {

    protected String maSach, tenSach, tacGia;
    protected int namXuatBan, soLuong;
    protected double donGia;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        sc.useLocale(Locale.US);

        ThucHanhTuan4 sach = new ThucHanhTuan4();
        sach.nhap(sc);

        // Bài 1: kiểm tra hợp lệ
        if (!sach.kiemTraHopLe()) {
            System.out.println("=> Sách KHÔNG hợp lệ. Dừng chương trình.");
            sc.close();
            return;
        }
        System.out.println("=> Sách hợp lệ.");

        String loaiGia = sach.phanLoaiGia();     // Bài 2
        String tonKho = sach.phanLoaiTonKho();   // Bài 3
        String loaiSach = chonLoaiSach(sc);      // Bài 4

        sach.hienThi(loaiGia, tonKho, loaiSach);
        sc.close();
    }

    protected void nhap(Scanner sc) {
        System.out.print("Nhập mã sách: ");
        maSach = sc.nextLine();
        System.out.print("Nhập tên sách: ");
        tenSach = sc.nextLine();
        System.out.print("Nhập tác giả: ");
        tacGia = sc.nextLine();
        System.out.print("Nhập năm xuất bản: ");
        namXuatBan = sc.nextInt();
        System.out.print("Nhập đơn giá: ");
        donGia = sc.nextDouble();
        System.out.print("Nhập số lượng: ");
        soLuong = sc.nextInt();
    }

    // Bài 1
    protected boolean kiemTraHopLe() {
        int namHienTai = java.time.Year.now().getValue();
        boolean hopLe = true;

        System.out.println("\n===== KIỂM TRA DỮ LIỆU =====");
        if (donGia <= 0) {
            System.out.println("Lỗi: Đơn giá phải lớn hơn 0!");
            hopLe = false;
        }
        if (soLuong < 0) {
            System.out.println("Lỗi: Số lượng phải >= 0!");
            hopLe = false;
        }
        if (namXuatBan <= 0 || namXuatBan > namHienTai) {
            System.out.println("Lỗi: Năm xuất bản phải từ 1 đến " + namHienTai + "!");
            hopLe = false;
        }
        return hopLe;
    }

    // Bài 2
    protected String phanLoaiGia() {
        if (donGia < 50000) {
            return "Sách giá thấp";
        } else if (donGia <= 100000) {
            return "Sách giá trung bình";
        } else if (donGia <= 200000) {
            return "Sách giá cao";
        } else {
            return "Sách giá đặc biệt";
        }
    }

    // Bài 3
    protected String phanLoaiTonKho() {
        if (soLuong == 0) {
            return "Hết sách";
        } else if (soLuong <= 5) {
            return "Sắp hết";
        } else if (soLuong <= 20) {
            return "Còn ít";
        } else {
            return "Còn nhiều";
        }
    }

    // Bài 4: switch
    protected static String chonLoaiSach(Scanner sc) {
        System.out.println("\n===== CHỌN LOẠI SÁCH =====");
        System.out.println("1. Sách giáo khoa");
        System.out.println("2. Sách tham khảo");
        System.out.println("3. Sách chuyên ngành");
        System.out.println("4. Sách khác");
        System.out.print("Nhập lựa chọn (1-4): ");
        int chon = sc.nextInt();

        switch (chon) {
            case 1:  return "Sách giáo khoa";
            case 2:  return "Sách tham khảo";
            case 3:  return "Sách chuyên ngành";
            case 4:  return "Sách khác";
            default: return "Không xác định (lựa chọn không hợp lệ)";
        }
    }

    protected void hienThi(String loaiGia, String tonKho, String loaiSach) {
        System.out.println("\n===== THÔNG TIN SÁCH =====");
        System.out.println("Mã sách       : " + maSach);
        System.out.println("Tên sách      : " + tenSach);
        System.out.println("Tác giả       : " + tacGia);
        System.out.println("Năm xuất bản  : " + namXuatBan);
        System.out.printf("Đơn giá       : %,.0f VNĐ%n", donGia);
        System.out.println("Số lượng      : " + soLuong);
        System.out.println("Phân loại giá : " + loaiGia);
        System.out.println("Tồn kho       : " + tonKho);
        System.out.println("Loại sách     : " + loaiSach);
    }
}
